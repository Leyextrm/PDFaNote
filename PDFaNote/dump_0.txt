140:         private int GetClosestLetterIndex(Point pt)
141:         {
142:             EnsureReadingOrderLetters();
143:             if (_readingOrderLetters.Count == 0) return -1;
144:             
145:             int bestIdx = -1;
146:             double bestDist = double.MaxValue;
147:             double scale = 2.0 / 0.75;
148:             
149:             for (int i = 0; i < _readingOrderLetters.Count; i++)
150:             {
151:                 var w = _readingOrderLetters[i];
152:                 double wLeft = w.Bounds.X * scale;
153:                 double wRight = wLeft + w.Bounds.Width * scale;
154:                 double wordTopPdf = w.Bounds.Y + w.Bounds.Height;
155:                 double wTop = InkCanvas.Height - (wordTopPdf * scale);
156:                 double wBottom = InkCanvas.Height - (w.Bounds.Y * scale);
157:                 
158:                 double cx = (wLeft + wRight) / 2.0;
159:                 double cy = (wTop + wBottom) / 2.0;
160:                 double dx = pt.X - cx;
161:                 double dy = pt.Y - cy;
162:                 double dist = dx*dx + dy*dy;
163:                 
164:                 if (dist < bestDist)

641:         {
642:             var pts = stroke.Points;
643:             for (int i = 1; i < pts.Count; i++)

646:             }

651:         private double GetDistance(Point a, Point b)
652:         {
653:             double dx = a.X - b.X;
654:             double dy = a.Y - b.Y;
655:             return Math.Sqrt(dx * dx + dy * dy);
656:         }

658:         private double DistanceToSegment(Point p, Point a, Point b)
659:         {
660:             double l2 = (a.X - b.X) * (a.X - b.X) + (a.Y - b.Y) * (a.Y - b.Y);
661:             if (l2 == 0) return GetDistance(p, a);
662:             double t = ((p.X - a.X) * (b.X - a.X) + (p.Y - a.Y) * (b.Y - a.Y)) / l2;
663:             t = Math.Max(0, Math.Min(1, t));
664:             Point proj = new Point(a.X + t * (b.X - a.X), a.Y + t * (b.Y - a.Y));
665:             return GetDistance(p, proj);
666:         }

689:         private bool IsPointInPolygon(Point p, Microsoft.UI.Xaml.Media.PointCollection poly)
690:         {
691:             Point p1, p2;
692:             bool inside = false;
693:             if (poly.Count < 3) return false;
694: 
695:             var oldPoint = new Point(poly[poly.Count - 1].X, poly[poly.Count - 1].Y);
696:             for (int i = 0; i < poly.Count; i++)
697:             {
698:                 var newPoint = new Point(poly[i].X, poly[i].Y);
699:                 if (newPoint.X > oldPoint.X) { p1 = oldPoint; p2 = newPoint; }
700:                 else { p1 = newPoint; p2 = oldPoint; }
701: 
702:                 if ((newPoint.X < p.X) == (p.X <= oldPoint.X)
703:                     && (p.Y - p1.Y) * (p2.X - p1.X) < (p2.Y - p1.Y) * (p.X - p1.X))
704:                 {
705:                     inside = !inside;
706:                 }
707:                 oldPoint = newPoint;
708:             }
709:             return inside;
710:         }
